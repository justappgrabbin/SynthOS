// App Registry - Tracks all pages/components that exist
class AppRegistry {
  static COMPONENTS_DIR = `${RNFS.DocumentDirectoryPath}/components`;
  static REGISTRY_FILE = `${RNFS.DocumentDirectoryPath}/registry.json`;

  // Initial seed app structure
  static SEED_STRUCTURE = {
    version: '1.0.0',
    pages: [
      {
        id: 'home',
        name: 'Home',
        type: 'screen',
        path: 'HomeScreen.js',
        enabled: true,
        createdAt: new Date().toISOString(),
        icon: '🏠'
      }
    ],
    components: [],
    navigation: {
      initialRoute: 'home',
      routes: ['home']
    }
  };

  static async initialize() {
    try {
      // Create components directory
      const exists = await RNFS.exists(this.COMPONENTS_DIR);
      if (!exists) {
        await RNFS.mkdir(this.COMPONENTS_DIR);
      }

      // Check if registry exists
      const registryExists = await RNFS.exists(this.REGISTRY_FILE);
      if (!registryExists) {
        await this.saveRegistry(this.SEED_STRUCTURE);
      }

      // Save initial home screen
      await this.saveComponent('home', this.getInitialHomeScreen());
    } catch (error) {
      console.error('Registry initialization failed:', error);
    }
  }

  static async getRegistry() {
    try {
      const registryData = await RNFS.readFile(this.REGISTRY_FILE, 'utf8');
      return JSON.parse(registryData);
    } catch (error) {
      return this.SEED_STRUCTURE;
    }
  }

  static async saveRegistry(registry) {
    try {
      await RNFS.writeFile(this.REGISTRY_FILE, JSON.stringify(registry, null, 2), 'utf8');
    } catch (error) {
      console.error('Failed to save registry:', error);
    }
  }

  static async addPage(pageDefinition, code) {
    try {
      const registry = await this.getRegistry();
      
      // Add page to registry
      const newPage = {
        id: pageDefinition.id,
        name: pageDefinition.name,
        type: 'screen',
        path: `${pageDefinition.id}.js`,
        enabled: true,
        createdAt: new Date().toISOString(),
        icon: pageDefinition.icon || '📄',
        evolutionId: pageDefinition.evolutionId
      };

      registry.pages.push(newPage);
      registry.navigation.routes.push(pageDefinition.id);
      registry.version = this.incrementVersion(registry.version);

      // Save component file
      await this.saveComponent(pageDefinition.id, code);
      
      // Update registry
      await this.saveRegistry(registry);

      return newPage;
    } catch (error) {
      console.error('Failed to add page:', error);
      throw error;
    }
  }

  static async addComponent(componentDef, code) {
    try {
      const registry = await this.getRegistry();
      
      const newComponent = {
        id: componentDef.id,
        name: componentDef.name,
        type: 'component',
        path: `${componentDef.id}.js`,
        enabled: true,
        createdAt: new Date().toISOString(),
        evolutionId: componentDef.evolutionId
      };

      registry.components.push(newComponent);
      
      await this.saveComponent(componentDef.id, code);
      await this.saveRegistry(registry);

      return newComponent;
    } catch (error) {
      console.error('Failed to add component:', error);
      throw error;
    }
  }

  static async saveComponent(componentId, code) {
    const filePath = `${this.COMPONENTS_DIR}/${componentId}.js`;
    await RNFS.writeFile(filePath, code, 'utf8');
  }

  static async getComponent(componentId) {
    try {
      const filePath = `${this.COMPONENTS_DIR}/${componentId}.js`;
      return await RNFS.readFile(filePath, 'utf8');
    } catch (error) {
      console.error(`Failed to load component ${componentId}:`, error);
      return null;
    }
  }

  static incrementVersion(version) {
    const [major, minor, patch] = version.split('.').map(Number);
    return `${major}.${minor}.${patch + 1}`;
  }

// Enhanced Evolution Engine that can generate entire new pages
class MultiPageEvolutionEngine {
  constructor(aiClient) {
    this.aiClient = aiClient;
    this.registry = null;
  }

  async initialize() {
    await AppRegistry.initialize();
    this.registry = await AppRegistry.getRegistry();
  }

  async analyzeAppStructure() {
    // Get full app structure
    const registry = await AppRegistry.getRegistry();
    
    // Load all current pages
    const pageContents = await Promise.all(
      registry.pages.map(async page => ({
        ...page,
        code: await AppRegistry.getComponent(page.id)
      }))
    );

    const analysisPrompt = `
Analyze this React Native app structure and suggest what new page or feature to add next:

CURRENT APP STRUCTURE:
${JSON.stringify(registry, null, 2)}

EXISTING PAGES:
${pageContents.map(p => `- ${p.name} (${p.id}): ${p.code?.substring(0, 200)}...`).join('\n')}

Suggest ONE new page/feature that would logically enhance this app.

Respond with ONLY valid JSON:
{
  "suggestedAddition": {
    "type": "page|component|feature",
    "id": "unique_id_for_this",
    "name": "Display Name",
    "description": "What this adds to the app",
    "benefit": "Why users would want this",
    "icon": "emoji",
    "complexity": 1-10,
    "priority": "high|medium|low"
  },
  "reasoning": "Why this is the logical next evolution"
}`;

    try {
      const analysis = await this.aiClient.generateAnalysis(analysisPrompt);
      return analysis;
    } catch (error) {
      console.error('App structure analysis failed:', error);
      throw error;
    }
  }

  async generateNewPage(pageDefinition) {
    const prompt = `
Generate a complete React Native screen component for this new page:

PAGE DEFINITION:
- ID: ${pageDefinition.id}
- Name: ${pageDefinition.name}
- Description: ${pageDefinition.description}
- Icon: ${pageDefinition.icon}

Requirements:
1. Must be a complete React Native functional component
2. Use hooks (useState, useEffect) as needed
3. Include proper imports
4. Create a complete StyleSheet
5. Make it visually appealing and user-friendly
6. Include the navigation prop parameter
7. Match the style of the existing app
8. Add meaningful functionality, not just placeholders

Generate ONLY the complete component code, no explanations.
`;

    try {
      const code = await this.aiClient.generateCode(prompt);
      return code;
    } catch (error) {
      console.error('Page generation failed:', error);
      throw error;
    }
  }

  async evolveNewPage(evolutionId) {
    try {
      // Step 1: Analyze what page to add
      const analysis = await this.analyzeAppStructure();
      const pageToAdd = analysis.suggestedAddition;

      // Step 2: Generate the new page code
      const pageCode = await this.generateNewPage(pageToAdd);

      // Step 3: Add to registry
      pageToAdd.evolutionId = evolutionId;
      await AppRegistry.addPage(pageToAdd, pageCode);

      // Reload registry
      this.registry = await AppRegistry.getRegistry();

      return {
        success: true,
        pageAdded: pageToAdd,
        code: pageCode,
        newVersion: this.registry.version
      };
    } catch (error) {
      console.error('Page evolution failed:', error);
      throw error;
    }
  }
}

// Simple Navigation Component
const SimpleNavigator = ({ registry, currentRoute, onNavigate }) => {
  if (!registry || !registry.pages) return null;

  return (
    <View style={navStyles.container}>
      <ScrollView 
        horizontal 
        showsHorizontalScrollIndicator={false}
        style={navStyles.scroll}
      >
        {registry.pages.map(page => (
          <TouchableOpacity
            key={page.id}
            style={[
              navStyles.navButton,
              currentRoute === page.id && navStyles.activeNavButton
            ]}
            onPress={() => onNavigate(page.id)}
          >
            <Text style={navStyles.navIcon}>{page.icon}</Text>
            <Text style={[
              navStyles.navText,
              currentRoute === page.id && navStyles.activeNavText
            ]}>
              {page.name}
            </Text>
          </TouchableOpacity>
        ))}
      </ScrollView>
    </View>
  );
};

// Dynamic Page Renderer
const DynamicPageRenderer = ({ pageId, registry }) => {
  const [pageComponent, setPageComponent] = useState(null);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    loadPage();
  }, [pageId]);

  const loadPage = async () => {
    setLoading(true);
    try {
      const code = await AppRegistry.getComponent(pageId);
      
      // In production, you'd use react-native-dynamic-code-eval or similar
      // For now, we'll show the code as a preview
      setPageComponent(code);
    } catch (error) {
      console.error('Failed to load page:', error);
    } finally {
      setLoading(false);
    }
  };

  if (loading) {
    return (
      <View style={styles.loadingContainer}>
        <ActivityIndicator size="large" color="#007AFF" />
        <Text style={styles.loadingText}>Loading {pageId}...</Text>
      </View>
    );
  }

  // In production, this would actually execute the component
  // For demo purposes, show info about the page
  const page = registry.pages.find(p => p.id === pageId);
  
  return (
    <View style={styles.pageContainer}>
      <Text style={styles.pageTitle}>{page.icon} {page.name}</Text>
      <Text style={styles.pageInfo}>Created: {new Date(page.createdAt).toLocaleDateString()}</Text>
      
      <View style={styles.codePreview}>
        <Text style={styles.codeTitle}>Component Code Preview:</Text>
        <ScrollView style={styles.codeScroll}>
          <Text style={styles.codeText}>{pageComponent?.substring(0, 500)}...</Text>
        </ScrollView>
      </View>

      <View style={styles.infoBox}>
        <Text style={styles.infoText}>
          ✨ In production, this would render the actual React Native component dynamically!
        </Text>
      </View>
    </View>
  );
};
